import tkinter as tk
from tkinter import ttk, messagebox
import csv
import os

FILE = "attendance.csv"

def add_student():
    name = name_entry.get()

    if not name:
        messagebox.showwarning("Warning", "Enter student name")
        return

    tree.insert("", "end", values=(name, "Present", "0%"))

    name_entry.delete(0, tk.END)


def mark_present():
    selected = tree.selection()

    if not selected:
        messagebox.showwarning("Warning", "Select a student")
        return

    for item in selected:
        values = list(tree.item(item)["values"])

        present = int(values[2].replace("%", ""))

        if present < 100:
            present += 10

        values[1] = "Present"
        values[2] = f"{present}%"

        tree.item(item, values=values)


def mark_absent():
    selected = tree.selection()

    if not selected:
        return

    for item in selected:
        values = list(tree.item(item)["values"])
        values[1] = "Absent"
        tree.item(item, values=values)


def save_data():
    with open(FILE, "w", newline="") as file:
        writer = csv.writer(file)

        writer.writerow(["Student", "Status", "Attendance"])

        for item in tree.get_children():
            writer.writerow(tree.item(item)["values"])

    messagebox.showinfo("Saved", "Attendance saved successfully!")


root = tk.Tk()
root.title("Student Attendance Manager")
root.geometry("750x500")

tk.Label(
    root,
    text="🎓 Student Attendance Manager",
    font=("Arial", 24, "bold")
).pack(pady=20)

frame = tk.Frame(root)
frame.pack()

name_entry = tk.Entry(frame, font=("Arial", 14), width=25)
name_entry.grid(row=0, column=0, padx=10)

tk.Button(
    frame,
    text="Add Student",
    command=add_student
).grid(row=0, column=1)

columns = ("Student", "Status", "Attendance")

tree = ttk.Treeview(
    root,
    columns=columns,
    show="headings"
)

for col in columns:
    tree.heading(col, text=col)

tree.pack(
    fill="both",
    expand=True,
    padx=30,
    pady=20
)

button_frame = tk.Frame(root)
button_frame.pack()

tk.Button(
    button_frame,
    text="✅ Present",
    command=mark_present
).grid(row=0, column=0, padx=10)

tk.Button(
    button_frame,
    text="❌ Absent",
    command=mark_absent
).grid(row=0, column=1, padx=10)

tk.Button(
    button_frame,
    text="💾 Save",
    command=save_data
).grid(row=0, column=2, padx=10)

root.mainloop()
